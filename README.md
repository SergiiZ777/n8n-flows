## 🔄 How to Run a Full Backup

Run this script **on your Hetzner server** (where n8n is hosted) to export all workflows as individual JSON files:

```bash
bash <(curl -s https://raw.githubusercontent.com/SergiiZ777/n8n-flows/main/scripts/backup.sh)
```

Or manually:

```bash
# Set your credentials
N8N_API_KEY="your-n8n-api-key"        # Settings → API → Create API Key
GITHUB_TOKEN="your-github-pat"
GITHUB_REPO="SergiiZ777/n8n-flows"
N8N_URL="http://localhost:5678"        # internal URL on the server

# Export and push all workflows
curl -s "$N8N_URL/api/v1/workflows?limit=200" \
  -H "X-N8N-API-KEY: $N8N_API_KEY" | \
  python3 -c "
import json, sys, subprocess, base64, urllib.request

data = json.load(sys.stdin)
token = '$GITHUB_TOKEN'
repo = '$GITHUB_REPO'

for wf in data['data']:
    wid = wf['id']
    # Get full workflow
    req = urllib.request.Request(
        f'$N8N_URL/api/v1/workflows/{wid}',
        headers={'X-N8N-API-KEY': '$N8N_API_KEY'}
    )
    full = json.loads(urllib.request.urlopen(req).read())
    name = wf['name'].replace('/', '_').replace(' ', '_')
    filename = f'workflows/{name}__{wid}.json'
    content = base64.b64encode(json.dumps(full, indent=2).encode()).decode()
    
    # Check if file exists (for SHA)
    try:
        chk = urllib.request.Request(
            f'https://api.github.com/repos/{repo}/contents/{filename}',
            headers={'Authorization': f'token {token}', 'Accept': 'application/vnd.github.v3+json'}
        )
        existing = json.loads(urllib.request.urlopen(chk).read())
        sha = existing.get('sha')
    except:
        sha = None
    
    body = {'message': f'backup: {wf[\"name\"]}', 'content': content, 'branch': 'main'}
    if sha:
        body['sha'] = sha
    
    put = urllib.request.Request(
        f'https://api.github.com/repos/{repo}/contents/{filename}',
        data=json.dumps(body).encode(),
        method='PUT',
        headers={'Authorization': f'token {token}', 'Content-Type': 'application/json'}
    )
    urllib.request.urlopen(put)
    print(f'✓ {wf[\"name\"]}')
"
```
